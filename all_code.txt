const canvasRenderer = L.canvas({ padding: 0.5 });

function MapSettings() {
  const map = useMap();

  useEffect(() => {
    if (!russiaGeoJson) return;

    const geoJsonLayer = L.geoJSON(russiaGeoJson);
    const bounds = geoJsonLayer.getBounds();
    
    // Р”РѕР±Р°РІР»СЏРµРј РѕС‚СЃС‚СѓРї 5%, С‡С‚РѕР±С‹ РїСЂРё РїРµСЂРµС‚Р°СЃРєРёРІР°РЅРёРё РєР°СЂС‚Р° РЅРµ СѓРїРёСЂР°Р»Р°СЃСЊ РІ СЃР°РјС‹Р№ РєСЂР°Р№
    const paddedBounds = bounds.pad(0.05);

    map.fitBounds(bounds, { padding: [20, 20] });
    map.setMaxBounds(paddedBounds);
    
    // РћРіСЂР°РЅРёС‡РёРІР°РµРј Р·СѓРј
    map.setMinZoom(map.getBoundsZoom(bounds));
    map.setMaxZoom(7);

    geoJsonLayer.remove();
  }, [map]);

  return null;
}

function CustomZoomControl() {
  const map = useMap();
  return (
    <div className="absolute top-4 left-4 z-[1000] flex flex-col bg-panel-dark border border-slate-700 rounded-lg overflow-hidden shadow-xl">
      <button onClick={() => map.zoomIn()} className="w-10 h-10 flex items-center justify-center text-xl font-bold text-slate-300 hover:text-white hover:bg-slate-700 transition-colors border-b border-slate-700">+</button>
      <button onClick={() => map.zoomOut()} className="w-10 h-10 flex items-center justify-center text-xl font-bold text-slate-300 hover:text-white hover:bg-slate-700 transition-colors">в€’</button>
    </div>
  );
}

export default function Map({ scenarioData, onRegionClick }) {
  const colorMap = getRegionColors(scenarioData);

  const styleFeature = (feature) => {
    const id = NAME_TO_ID[feature.properties.name];
    
    if (id && colorMap[id]) {
      return {
        fillColor: colorMap[id],
        weight: 1,
        opacity: 1,
        color: '#475569',
        fillOpacity: 0.9
      };
    }

    return {
      fillColor: '#1e293b',
      weight: 1,
      opacity: 1,
      color: '#475569',
      fillOpacity: 1
    };
  };

  const onEachFeature = (feature, layer) => {
    const id = NAME_TO_ID[feature.properties.name];
    
    if (id) {
      layer.on({
        mouseover: (e) => {
          const layer = e.target;
          layer.setStyle({ weight: 2, color: '#ffffff' });
        },
        mouseout: (e) => {
          const layer = e.target;
          layer.setStyle(styleFeature(feature));
        },
        click: () => {
          onRegionClick(id);
        }
      });
    }
  };

  return (
    <div style={{ height: '100%', width: '100%', position: 'relative' }}>
      <MapContainer 
        center={[62, 95]} 
        zoom={3} 
        style={{ height: '100%', width: '100%', position: 'absolute', top: 0, left: 0, zIndex: 1, backgroundColor: '#0f172a' }}
        zoomControl={false}
        attributionControl={false}
        maxBoundsViscosity={1.0}
        preferCanvas={true}
        zoomSnap={0.1}
        // РћРўРљР›Р®Р§РђР•Рњ РђРќРРњРђР¦РР Р”Р›РЇ РџР›РђР’РќРћРЎРўР:
        zoomAnimation={false} 
        fadeAnimation={false}
        inertia={false} 
      >
        <MapSettings />
        <CustomZoomControl />
        
        <GeoJSON 
          key={JSON.stringify(colorMap)} 
          data={russiaGeoJson} 
          style={styleFeature} 
          onEachFeature={onEachFeature}
          renderer={canvasRenderer} // РџР•Р Р•Р”РђР•Рњ РќРђРЁ Р‘РЈР¤Р•Р РќР«Р™ Р Р•РќР”Р•Р Р•Р 
        />
      </MapContainer>
    </div>
  );
}

// Р’РѕР·РІСЂР°С‰Р°РµС‚ РѕР±СЉРµРєС‚ { regionId: color }
export function getRegionColors(scenarioData) {
  const entries = Object.entries(scenarioData || {});
  // РЎРѕСЂС‚РёСЂСѓРµРј РїРѕ СѓР±С‹РІР°РЅРёСЋ (1 РјРµСЃС‚Рѕ = СЃР°РјРѕРµ Р±РѕР»СЊС€РѕРµ Р·РЅР°С‡РµРЅРёРµ)
  entries.sort((a, b) => b[1] - a[1]);

  const colorMap = {};
  entries.forEach(([id], index) => {
    colorMap[id] = RANK_COLORS[index] || RANK_COLORS[RANK_COLORS.length - 1];
  });
  return colorMap;
}

// Р’РѕР·РІСЂР°С‰Р°РµС‚ РјР°СЃСЃРёРІ РѕС‚СЃРѕСЂС‚РёСЂРѕРІР°РЅРЅС‹С… СЂРµРіРёРѕРЅРѕРІ РґР»СЏ С‚Р°Р±Р»РёС†С‹
export function getSortedRegions(scenarioData) {
  return Object.entries(scenarioData || {})
    .sort((a, b) => b[1] - a[1])
    .map(([id, value], index) => ({ id, value, rank: index + 1 }));
}
